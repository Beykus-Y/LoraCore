-- =======================================================
-- LoraCore ModuOS Installer v5.0 (Corrected Boot Script)
-- =======================================================
local term = tablet.terminal
local gpu = tablet.gpu
local fs = _G.fs
local colors = _G.colors
local os = _G.os

term.clear()
gpu.setBackgroundColor(colors.black)
gpu.setTextColor(colors.white)
term.setCursorPos(1, 1)

term.print("Starting ModuOS Installation...")
term.print("-------------------------------")

-- ИСПРАВЛЕННЫЙ BOOT SCRIPT
local boot_script_content = [[
-- /boot.lua - Minimal Operating System v1.2 (Persistent)
local term = tablet.terminal

term.clear()
term.setCursorPos(1, 1)
print("ModuOS v0.2 Loaded.")
print("Type 'help' for commands.")

-- Главный цикл ОС
while true do
    term.write("> ")
    local cmd = term.read()
    if cmd == "help" then
        print("Commands: help, reboot, exit")
    elseif cmd == "reboot" then
        os.reboot()
        break -- Выходим из цикла, чтобы BIOS перезагрузился
    elseif cmd == "exit" then
        term.clear()
        term.print("System halted. Press any key to reboot.")
        os.pullEvent("key") -- Ждем нажатия клавиши
        os.reboot()
        break -- Выходим из цикла
    else
        print("Unknown command: " .. tostring(cmd))
    end
end
]]

-- Процесс установки
term.print("\n1. Writing boot file... ")
fs.write("/boot.lua", boot_script_content)
term.print("[OK]")

term.print("2. Creating directories... ")
fs.makeDir("/usr")
fs.makeDir("/home")
term.print("[OK]")

term.print("3. Verifying installation... ")
local content = fs.read("/boot.lua")
if not content then
    gpu.setTextColor(colors.red)
    term.print("\n   ERROR: Verification failed! Could not read boot file back.")
    return false
end
term.print("[OK]")

return true
