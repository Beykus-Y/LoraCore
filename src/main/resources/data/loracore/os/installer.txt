-- =======================================================
-- LoraCoreOS Lua Installer v2.0 (Graphical, Non-Interactive)
-- =======================================================
-- Этот скрипт вызывается из recovery.lua и не требует
-- пользовательского ввода. Он возвращает true при успехе.
-- =======================================================

local gpu = tablet.gpu
local fs = _G.fs
local os = _G.os
local colors = _G.colors

local W, H = 480, 270
local current_line = 4

-- Вспомогательная функция для вывода статуса
local function print_status(message, status)
    local y = current_line * 12
    gpu.drawText(25, y, message, colors.white)

    local status_text = status and "[ OK ]" or "[FAIL]"
    local status_color = status and colors.lime or colors.red
    local status_x = W - 100
    gpu.drawText(status_x, y, status_text, status_color)

    current_line = current_line + 1
    os.sleep(0.3)
end

-- Код самого boot.lua, который будет установлен на диск
local shell_code = [[
-- /boot/boot.lua - LoraCore Simple Shell
local term = tablet.terminal
local fs = _G.fs
local current_path = "/"
_G.programs = {}
function _G.programs.help() print("Commands: help, ls, cat, reboot, clear, run") end
function _G.programs.ls(a) local p=a[1]or current_path;local f,e=fs.list(p);if not f then print("ls: "..tostring(e))return end;for i,v in ipairs(f)do print(v)end end
function _G.programs.cat(a) local p=a[1];if not p then print("Usage: cat <file>")return end;local c,e=fs.read(p);if not c then print("cat: "..tostring(e))return end;print(c)end
function _G.programs.run(a) local p=a[1];if not p then print("Usage: run <f.lua>")return end;local c,e=fs.read(p);if not c then print("run: "..tostring(e))return end;local f,l=load(c,"@"..p,"t",_G);if not f then print("Syntax error: "..tostring(l))return end;local s,r=pcall(f);if not s then print("Runtime error: "..tostring(r))end end
term.clear();print("LoraCore Lua Shell v1.1. Type 'help'.")
while true do term.write(current_path.."> ");local l=term.read();if l and #l>0 then local p={};for v in l:gmatch("[^%s]+")do table.insert(p,v)end;local c=table.remove(p,1);if c=="reboot"then os.reboot()elseif c=="clear"then term.clear()elseif _G.programs[c]then _G.programs[c](p)else print("Not found: "..c)end end end
]]

-- Основная функция установщика
local function install()
    -- 1. Очистка диска (простая версия)
    gpu.drawText(25, current_line * 12, "Formatting disk...", colors.white)
    current_line = current_line + 1
    os.sleep(0.5)

    local files, err = fs.list("/")
    if files then
        for i, file in ipairs(files) do
            fs.delete(file)
        end
    end
    print_status("Disk formatted.", true)

    -- 2. Создание директорий
    local dir_boot_ok = fs.makeDir("/boot") -- ИСПРАВЛЕНО: Создаем папку /boot
    print_status("Creating /boot...", dir_boot_ok)
    if not dir_boot_ok then return false end
    local dir1_ok = fs.makeDir("/bin")
    print_status("Creating /bin...", dir1_ok)
    if not dir1_ok then return false end

    local dir2_ok = fs.makeDir("/etc")
    print_status("Creating /etc...", dir2_ok)
    if not dir2_ok then return false end

    local dir3_ok = fs.makeDir("/home")
    print_status("Creating /home...", dir3_ok)
    if not dir3_ok then return false end

    -- 3. Запись загрузочного скрипта
    current_line = current_line + 1
    gpu.drawText(25, current_line * 12, "Writing bootloader...", colors.white)
    current_line = current_line + 1

    local write_ok = fs.write("/boot/boot.lua", shell_code)
    print_status("Writing /boot/boot.lua...", write_ok)
    if not write_ok then return false end

    -- 4. Проверка
    local verify_ok = fs.exists("/boot/boot.lua")
    print_status("Verifying bootloader...", verify_ok)
    if not verify_ok then return false end

    -- Если все прошло успешно, возвращаем true
    return true
end

-- Запускаем установку и возвращаем результат
return install()
